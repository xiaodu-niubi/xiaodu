# 第8课：大模型客户端 —— 让代码能调用AI大脑

> 🎯 **本课目标**：封装DeepSeek API调用，让所有智能体都能方便地和AI模型对话。

---

## 8.1 我们要封装什么？

项目中所有智能体都需要和大模型（DeepSeek API）交流。如果每个智能体都写一遍API调用代码，就会有很多重复代码。

所以我们要写一个**统一的LLM客户端**：
- `build_llm()` → 创建一个大模型实例（支持流式输出 + 工具绑定）
- `format_messages_for_llm()` → 把对话历史组装成大模型认识的格式

---

## 8.2 理解LangChain的ChatOpenAI

DeepSeek的API和OpenAI的API**格式兼容**，所以我们可以用 `langchain-openai` 这个包来调用DeepSeek：

```python
from langchain_openai import ChatOpenAI

llm = ChatOpenAI(
    model="deepseek-v4-pro",          # 模型名
    api_key="sk-xxx",                 # API密钥
    base_url="https://api.deepseek.com/v1",  # ⭐ 关键：指向DeepSeek而非OpenAI
)
```

## 8.3 先写数据模型：`src/models/schemas.py`

这个是API请求和响应的数据格式定义，用Pydantic库来校验数据。

```python
"""Pydantic 数据模型 - API 请求/响应模式定义"""
from datetime import datetime
from typing import Optional

from pydantic import BaseModel, Field


# ═══════════════════════════════════════════
# 聊天相关
# ═══════════════════════════════════════════

class ChatRequest(BaseModel):
    """聊天请求的数据格式"""
    message: str                                    # 必填：用户说的话
    conversation_id: Optional[str] = None           # 可选：已有的对话ID
    stream: bool = True                             # 是否流式输出


class ChatResponse(BaseModel):
    """聊天响应的数据格式"""
    conversation_id: str
    session_id: str
    message: str
    tool_calls: Optional[list[dict]] = None


class ChatStreamChunk(BaseModel):
    """流式响应的一小块"""
    type: str  # 类型：'token'(文字), 'tool_call'(工具调用), 'tool_result'(工具结果), 'done'(结束)
    content: str = ""
    metadata: Optional[dict] = None


# ═══════════════════════════════════════════
# 对话管理
# ═══════════════════════════════════════════

class ConversationCreate(BaseModel):
    """创建对话的请求"""
    title: str = "新对话"


class ConversationUpdate(BaseModel):
    """更新对话标题的请求"""
    title: str


class MessageItem(BaseModel):
    """一条消息的数据格式"""
    id: int
    role: str
    content: Optional[str]
    tool_calls: Optional[list] = None
    metadata: Optional[dict] = None
    created_at: datetime


class ConversationItem(BaseModel):
    """对话列表中的一项"""
    id: str
    session_id: str
    title: str
    status: str
    created_at: datetime
    updated_at: datetime


class ConversationDetail(BaseModel):
    """对话详情（含消息列表）"""
    id: str
    session_id: str
    title: str
    status: str
    created_at: datetime
    updated_at: datetime
    messages: list[MessageItem]


# ═══════════════════════════════════════════
# 知识库
# ═══════════════════════════════════════════

class KnowledgeDocument(BaseModel):
    """知识库文档"""
    id: str
    filename: str
    file_type: str
    chunk_count: int
    status: str
    created_at: datetime


class KnowledgeUploadResponse(BaseModel):
    """文档上传响应"""
    id: str
    filename: str
    chunk_count: int
    status: str


# ═══════════════════════════════════════════
# 通用
# ═══════════════════════════════════════════

class ErrorResponse(BaseModel):
    """错误响应"""
    error: str
    detail: Optional[str] = None
```

> 💡 **Pydantic 是什么？**
>
> Pydantic 是一个数据校验库。它可以：
> 1. 自动校验数据类型（比如 `message: str` 就会确保传进来的是字符串）
> 2. 自动生成JSON Schema文档
> 3. FastAPI直接用它做请求参数校验
>
> 如果没有Pydantic，你需要手写一大堆 `if not isinstance(...)` 的代码。

## 8.4 写代码：`src/models/llm_client.py`

这是**核心中的核心**——大模型调用的封装。

```python
"""大模型客户端 - 兼容 OpenAI/DeepSeek API"""
from typing import Optional

from langchain_openai import ChatOpenAI
from langchain_core.messages import (
    BaseMessage, HumanMessage, SystemMessage, AIMessage, ToolMessage
)
from langchain_core.tools import BaseTool

from src.config import settings


def build_llm(
    temperature: Optional[float] = None,
    streaming: bool = True,
    tools: Optional[list] = None,
) -> ChatOpenAI:
    """构建 LangChain 兼容的 LLM 实例，指向 DeepSeek API
    
    参数：
    - temperature: 创造性参数（0=保守，1=天马行空），不传则用配置文件里的默认值
    - streaming: 是否流式输出（True=打字机效果，False=等全部生成完再返回）
    - tools: 可以绑定的工具列表（让LLM能调用函数）
    
    返回：一个配置好的 ChatOpenAI 实例
    """
    # 基础参数
    kwargs = dict(
        model=settings.llm_model,           # 模型名称
        api_key=settings.llm_api_key,       # API密钥
        base_url=settings.llm_base_url,     # DeepSeek API地址
        temperature=temperature or settings.llm_temperature,
        max_tokens=settings.llm_max_tokens,
        streaming=streaming,
    )

    llm = ChatOpenAI(**kwargs)

    # 如果有工具，绑定到LLM
    # bind_tools 告诉LLM："你可以调用这些函数来获取更多信息"
    if tools:
        converted = []
        for t in tools:
            if isinstance(t, dict):
                # 已经是OpenAI格式的字典，直接用
                converted.append(t)
            elif isinstance(t, BaseTool):
                # LangChain的BaseTool实例，转换格式
                converted.append(_base_tool_to_openai_schema(t))
            elif callable(t):
                # 原生Python函数，转换格式
                converted.append(_function_to_openai_schema(t))
        if converted:
            llm = llm.bind_tools(converted)

    return llm


def _function_to_openai_schema(func) -> dict:
    """将 Python 函数转换为 OpenAI 工具定义模式
    
    例如 calculator(expression: str) 这个函数会被转成：
    {
        "type": "function",
        "function": {
            "name": "calculator",
            "description": "安全地计算数学表达式...",
            "parameters": {
                "type": "object",
                "properties": {
                    "expression": {"type": "string", "description": "参数：expression"}
                },
                "required": ["expression"]
            }
        }
    }
    """
    import inspect
    sig = inspect.signature(func)
    doc = (func.__doc__ or func.__name__).strip()

    props = {}
    required = []
    for name, param in sig.parameters.items():
        if name in ("self", "cls"):
            continue
        # 映射Python类型到JSON Schema类型
        param_type = "string"
        if param.annotation is not inspect.Parameter.empty:
            anno = param.annotation
            type_map = {int: "integer", float: "number", bool: "boolean", str: "string"}
            param_type = type_map.get(anno, "string")
        props[name] = {"type": param_type, "description": f"参数：{name}"}
        if param.default is inspect.Parameter.empty:
            required.append(name)

    return {
        "type": "function",
        "function": {
            "name": func.__name__,
            "description": doc,
            "parameters": {
                "type": "object",
                "properties": props,
                "required": required,
            },
        },
    }


def _base_tool_to_openai_schema(t: BaseTool) -> dict:
    """将 LangChain BaseTool 实例转换为 OpenAI 工具定义模式"""
    params = {"type": "object", "properties": {}}
    if hasattr(t, 'args_schema') and t.args_schema:
        try:
            params = t.args_schema.model_json_schema()
        except Exception:
            pass

    return {
        "type": "function",
        "function": {
            "name": t.name,
            "description": t.description or "",
            "parameters": params,
        },
    }


def format_messages_for_llm(
    system_prompt: str,
    history: list[dict],
    user_message: str,
) -> list[BaseMessage]:
    """将系统提示、历史记录和当前消息组装为 LLM 消息列表
    
    大模型的对话格式是：
    [
        SystemMessage("你是一个有用的助手"),  ← 系统提示（定义角色和行为）
        HumanMessage("你好"),                 ← 用户第1轮
        AIMessage("你好！有什么可以帮你？"),    ← 助手第1轮
        HumanMessage("天气怎么样"),           ← 用户第2轮（当前问题）
    ]
    """
    messages: list[BaseMessage] = [SystemMessage(content=system_prompt)]

    # 添加历史消息（最多 max_context_messages 条，防止token超限）
    for msg in history[-settings.max_context_messages:]:
        role = msg.get("role", "user")
        content = msg.get("content", "")
        if role == "user":
            messages.append(HumanMessage(content=content))
        elif role == "assistant":
            messages.append(AIMessage(content=content))
        elif role == "system":
            messages.append(SystemMessage(content=content))

    # 最后添加当前用户消息
    messages.append(HumanMessage(content=user_message))
    return messages


def parse_tool_calls(response) -> list[dict]:
    """从 LLM 响应中提取工具调用信息"""
    tool_calls = []
    if hasattr(response, "tool_calls") and response.tool_calls:
        for tc in response.tool_calls:
            tool_calls.append({
                "name": tc.get("name", ""),
                "args": tc.get("args", {}),
                "id": tc.get("id", ""),
            })
    return tool_calls
```

## 8.5 关键概念：流式输出 vs 非流式输出

```
非流式（streaming=False）：
  发送请求 → 等待...→ 一次性收到完整回复
  "请"
  "请问"
  "请问有"
  ...（内部生成中，你看不到）
  "请问有什么可以帮你的？" ← 全部生成完才给你

流式（streaming=True）：
  发送请求 → 一个字一个字地收到
  "请" → UI显示"请"
  "问" → UI显示"请问"
  "有" → UI显示"请问有"
  "什么可以帮你的？" → UI显示完整回复
```

流式输出的用户体验更好（像打字机效果），我们的聊天接口默认用流式。

## 8.6 关键概念：工具绑定（bind_tools）

```python
# 不给工具：LLM只能纯文字回复
llm = build_llm()

# 给工具：LLM可以"调用函数"
llm_with_tools = build_llm(tools=[calculator, web_search])

# 当用户问"1+1等于几"，LLM会回复：
# "我需要调用calculator工具来计算" ← LLM不会自己算
# 然后返回一个函数调用：calculator(expression="1+1")
# 你的代码执行这个函数，把结果"2"告诉LLM
# LLM回复："1+1等于2"  ← 基于工具结果回答
```

## 8.7 测试LLM客户端

创建 `test_llm.py`（确保 `.env` 中填了真实的API密钥）：

```python
"""测试大模型客户端"""
from src.models.llm_client import build_llm, format_messages_for_llm

print("=" * 50)
print("测试LLM客户端")
print("=" * 50)

# 测试1：构建LLM实例
print("\n1. 构建LLM实例...")
llm = build_llm()
print("   ✅ LLM实例创建成功")

# 测试2：组装消息
print("\n2. 测试消息组装...")
messages = format_messages_for_llm(
    system_prompt="你是一个有用的助手，用中文简短回答。",
    history=[
        {"role": "user", "content": "你好"},
        {"role": "assistant", "content": "你好！有什么可以帮你的？"},
    ],
    user_message="1+1等于几？",
)
print(f"   ✅ 共 {len(messages)} 条消息")
print(f"   第一条：{messages[0]}")
print(f"   最后一条：{messages[-1]}")

# 测试3：调用LLM（非流式）
print("\n3. 调用LLM（非流式）...")
response = llm.invoke(messages)
content = response.content if hasattr(response, "content") else str(response)
print(f"   LLM回复：{content}")

# 测试4：流式调用
print("\n4. 调用LLM（流式）...")
llm_stream = build_llm(streaming=True)
print("   ", end="")
for chunk in llm_stream.stream("用一句话介绍Python语言"):
    if chunk.content:
        print(chunk.content, end="", flush=True)
print()

print("\n✅ 所有测试完成！")
```
